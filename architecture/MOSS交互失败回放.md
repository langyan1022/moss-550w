## 回放命令

每条命令使用隔离数据目录并生成新的 `prototype/runs` 报告。浏览器、文件麦克风和 CUDA 用项目的硬件测试脚本串行运行。第一组点歌与天气回放会读取实际 G 盘索引和公开天气数据。只回放点歌时给第一条命令加 `--music-only`。

$repo = '.'
$proto = "$repo/prototype"

function Invoke-MossReplay([string]$Name, [string]$Script, [string[]]$Flags = @(), [int]$MinFreeGiB = 6) {
    & "$repo/run-hardware-benchmark.ps1" -Name $Name -Executable "$proto/.venv/Scripts/python.exe" `
        -TestArguments (@('-X', 'utf8', '-B', $Script) + $Flags) -WorkDirectory $proto -MinFreeGiB $MinFreeGiB
    if ($LASTEXITCODE) { throw "回放失败或未启动 $Name，退出码 $LASTEXITCODE" }
}
